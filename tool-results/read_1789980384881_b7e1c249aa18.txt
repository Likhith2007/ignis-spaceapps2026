    38→function hexbin(points: { lon: number; lat: number; frp: number }[], bbox: number[]) {
    39→  const span = Math.max(bbox[2] - bbox[0], bbox[3] - bbox[1]);
    40→  const bin = Math.min(2, Math.max(0.25, +(span / 24).toFixed(2)));
    41→  const map = new Map<string, { n: number; frp: number; lat: number; lon: number }>();
    42→  for (const p of points) {
    43→    const gx = Math.floor(p.lon / bin), gy = Math.floor(p.lat / bin);
    44→    const k = `${gx},${gy}`;
    45→    const g = map.get(k) || { n: 0, frp: 0, lat: (gy + 0.5) * bin, lon: (gx + 0.5) * bin };
    46→    g.n++; g.frp += p.frp || 0;
    47→    map.set(k, g);
    48→  }
    49→  const maxN = Math.max(1, ...[...map.values()].map((v) => v.n));
    50→  return { bin, maxN, features: [...map.values()] };
    51→}
